package example.generated
private val trevrpcCodec0: zip.trev.trevrpc.MessageCodec<example.generated.HelloRequest> =
        decode = { body -> example.generated.HelloRequest.parseFrom(body) },
private val trevrpcCodec1: zip.trev.trevrpc.MessageCodec<example.generated.HelloReply> =
        decode = { body -> example.generated.HelloReply.parseFrom(body) },

public const val GREETER_SERVICE_NAME: String = "hello.v1.Greeter"
public const val GREETER_SAY_HELLO_METHOD_NAME: String = "SayHello"
public val GREETER_SAY_HELLO_RPC_KIND: zip.trev.trevrpc.RpcKind = zip.trev.trevrpc.RpcKind.UNARY
public const val GREETER_LOTS_OF_REPLIES_METHOD_NAME: String = "LotsOfReplies"
public val GREETER_LOTS_OF_REPLIES_RPC_KIND: zip.trev.trevrpc.RpcKind = zip.trev.trevrpc.RpcKind.SERVER_STREAMING
public const val GREETER_LOTS_OF_GREETINGS_METHOD_NAME: String = "LotsOfGreetings"
public val GREETER_LOTS_OF_GREETINGS_RPC_KIND: zip.trev.trevrpc.RpcKind = zip.trev.trevrpc.RpcKind.CLIENT_STREAMING
public const val GREETER_BIDI_HELLO_METHOD_NAME: String = "BidiHello"
public val GREETER_BIDI_HELLO_RPC_KIND: zip.trev.trevrpc.RpcKind = zip.trev.trevrpc.RpcKind.BIDIRECTIONAL_STREAMING
public interface GreeterService {
    public suspend fun sayHello(
    public suspend fun lotsOfReplies(
    ): Flow<example.generated.HelloReply>
    public suspend fun lotsOfGreetings(
        requests: Flow<example.generated.HelloRequest>,
    public suspend fun bidiHello(
        requests: Flow<example.generated.HelloRequest>,
    ): Flow<example.generated.HelloReply>
public class GreeterClient(
    public suspend fun sayHelloEnvelope(

    public suspend fun lotsOfRepliesCall(

public fun registerGreeter(
    server.routeUnary(GREETER_SERVICE_NAME, GREETER_SAY_HELLO_METHOD_NAME) { context, body ->
        requireTrevrpcKind(context, GREETER_SAY_HELLO_RPC_KIND)
        val request = decodeTrevrpcRequest(trevrpcCodec0, body)
        zip.trev.trevrpc.ResponseEnvelope(trevrpcCodec1.encode(service.sayHello(context, request)))
    server.routeServerStreaming(GREETER_SERVICE_NAME, GREETER_LOTS_OF_REPLIES_METHOD_NAME) { context, body ->
        requireTrevrpcKind(context, GREETER_LOTS_OF_REPLIES_RPC_KIND)
        val request = decodeTrevrpcRequest(trevrpcCodec0, body)
        zip.trev.trevrpc.ResponseEnvelope(service.lotsOfReplies(context, request).map(trevrpcCodec1.encode))
    server.routeClientStreaming(GREETER_SERVICE_NAME, GREETER_LOTS_OF_GREETINGS_METHOD_NAME) { context, requests ->
        requireTrevrpcKind(context, GREETER_LOTS_OF_GREETINGS_RPC_KIND)
        val decoded = requests.map { body -> decodeTrevrpcRequest(trevrpcCodec0, body) }
        zip.trev.trevrpc.ResponseEnvelope(trevrpcCodec1.encode(service.lotsOfGreetings(context, decoded)))
    server.routeBidirectionalStreaming(GREETER_SERVICE_NAME, GREETER_BIDI_HELLO_METHOD_NAME) { context, requests ->
        requireTrevrpcKind(context, GREETER_BIDI_HELLO_RPC_KIND)
        val decoded = requests.map { body -> decodeTrevrpcRequest(trevrpcCodec0, body) }
        zip.trev.trevrpc.ResponseEnvelope(service.bidiHello(context, decoded).map(trevrpcCodec1.encode))
