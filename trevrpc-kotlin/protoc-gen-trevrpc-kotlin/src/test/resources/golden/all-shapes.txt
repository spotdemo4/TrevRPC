package example.generated
private val trevrpcCodec0: zip.trev.trevrpc.MessageCodec<example.generated.HelloRequest> =
        decode = { body -> zip.trev.trevrpc.decodeProtobuf(body, example.generated.HelloRequest.getDescriptor(), example.generated.HelloRequest::parseFrom) },
        messageType = "hello.v1.HelloRequest",
private val trevrpcCodec1: zip.trev.trevrpc.MessageCodec<example.generated.HelloReply> =
        decode = { body -> zip.trev.trevrpc.decodeProtobuf(body, example.generated.HelloReply.getDescriptor(), example.generated.HelloReply::parseFrom) },
        messageType = "hello.v1.HelloReply",
public const val GREETER_SAY_HELLO_PATH: String = "/hello.v1.Greeter/SayHello"
public const val GREETER_LOTS_OF_REPLIES_PATH: String = "/hello.v1.Greeter/LotsOfReplies"
public const val GREETER_LOTS_OF_GREETINGS_PATH: String = "/hello.v1.Greeter/LotsOfGreetings"
public const val GREETER_BIDI_HELLO_PATH: String = "/hello.v1.Greeter/BidiHello"
public class GreeterClient(
    public suspend fun sayHello(
        options: zip.trev.trevrpc.CallOptions.() -> zip.trev.trevrpc.CallOptions = { this },
        client.unary(GREETER_SAY_HELLO_PATH, request, trevrpcCodec0, trevrpcCodec1, defaultCallOptions.options())
    public fun lotsOfReplies(
        options: zip.trev.trevrpc.CallOptions.() -> zip.trev.trevrpc.CallOptions = { this },

        client.serverStreaming(GREETER_LOTS_OF_REPLIES_PATH, request, trevrpcCodec0, trevrpcCodec1, defaultCallOptions.options())

        requests: Flow<example.generated.HelloRequest>,
        options: zip.trev.trevrpc.CallOptions.() -> zip.trev.trevrpc.CallOptions = { this },
        client.clientStreaming(GREETER_LOTS_OF_GREETINGS_PATH, requests, trevrpcCodec0, trevrpcCodec1, defaultCallOptions.options())

        requests: Flow<example.generated.HelloRequest>,
        options: zip.trev.trevrpc.CallOptions.() -> zip.trev.trevrpc.CallOptions = { this },
    ): Flow<example.generated.HelloReply> =
        client.bidirectionalStreaming(GREETER_BIDI_HELLO_PATH, requests, trevrpcCodec0, trevrpcCodec1, defaultCallOptions.options())
